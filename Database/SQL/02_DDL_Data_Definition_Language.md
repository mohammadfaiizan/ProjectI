## 1. CREATE TABLE

### Basic Syntax
```sql
CREATE TABLE table_name (
    column1 datatype [constraints],
    column2 datatype [constraints],
    ...
    [table_constraints]
);
```

### Simple Example
```sql
CREATE TABLE employees (
    id          INT             NOT NULL AUTO_INCREMENT,
    first_name  VARCHAR(50)     NOT NULL,
    last_name   VARCHAR(50)     NOT NULL,
    email       VARCHAR(100)    UNIQUE,
    salary      DECIMAL(10, 2)  DEFAULT 0.00,
    dept_id     INT,
    hire_date   DATE            DEFAULT CURRENT_DATE,
    is_active   BOOLEAN         DEFAULT TRUE,
    PRIMARY KEY (id),
    FOREIGN KEY (dept_id) REFERENCES departments(id)
);
```

### With All Constraint Types
```sql
CREATE TABLE orders (
    order_id     INT            NOT NULL,
    customer_id  INT            NOT NULL,
    product_id   INT            NOT NULL,
    quantity     INT            NOT NULL CHECK (quantity > 0),
    unit_price   DECIMAL(10,2)  NOT NULL CHECK (unit_price >= 0),
    status       VARCHAR(20)    NOT NULL DEFAULT 'pending',
    order_date   TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    notes        TEXT,

    -- Table-level constraints
    CONSTRAINT pk_orders       PRIMARY KEY (order_id),
    CONSTRAINT fk_orders_cust  FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE CASCADE,
    CONSTRAINT fk_orders_prod  FOREIGN KEY (product_id)  REFERENCES products(id)  ON DELETE RESTRICT,
    CONSTRAINT chk_status      CHECK (status IN ('pending', 'processing', 'shipped', 'delivered', 'cancelled'))
);
```

### CREATE TABLE IF NOT EXISTS
```sql
CREATE TABLE IF NOT EXISTS products (
    id    INT          PRIMARY KEY AUTO_INCREMENT,
    name  VARCHAR(100) NOT NULL,
    price DECIMAL(10,2)
);
```

### CREATE TABLE AS (Copy from SELECT)
```sql
-- Create table from query result (copies data + structure, NOT constraints)
CREATE TABLE employees_backup AS
SELECT * FROM employees WHERE is_active = TRUE;

-- Create empty table with same structure
CREATE TABLE employees_archive AS
SELECT * FROM employees WHERE 1 = 0;

-- PostgreSQL equivalent
CREATE TABLE employees_backup AS TABLE employees;
```

### CREATE TABLE LIKE (MySQL — copy structure only)
```sql
CREATE TABLE employees_archive LIKE employees;
```

### PostgreSQL: INHERITS
```sql
CREATE TABLE managers (
    department VARCHAR(100)
) INHERITS (employees);
```

### PostgreSQL: Generated Columns
```sql
CREATE TABLE rectangles (
    width  FLOAT NOT NULL,
    height FLOAT NOT NULL,
    area   FLOAT GENERATED ALWAYS AS (width * height) STORED
);
```

### Auto-Increment Syntax by Database

```sql
-- MySQL
id INT AUTO_INCREMENT PRIMARY KEY

-- PostgreSQL
id SERIAL PRIMARY KEY
id BIGSERIAL PRIMARY KEY
id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY
id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY

-- SQL Server
id INT IDENTITY(1,1) PRIMARY KEY

-- SQLite
id INTEGER PRIMARY KEY AUTOINCREMENT
```

## 2. ALTER TABLE

### Add Column
```sql
ALTER TABLE table_name
ADD COLUMN column_name datatype [constraints];

-- Example
ALTER TABLE employees
ADD COLUMN phone VARCHAR(20);

ALTER TABLE employees
ADD COLUMN bonus DECIMAL(10,2) DEFAULT 0.00 NOT NULL;

-- Add multiple columns (MySQL / PostgreSQL)
ALTER TABLE employees
ADD COLUMN middle_name VARCHAR(50),
ADD COLUMN linkedin_url VARCHAR(200);
```

### Drop Column
```sql
ALTER TABLE table_name
DROP COLUMN column_name;

-- Example
ALTER TABLE employees DROP COLUMN middle_name;

-- Drop multiple columns
ALTER TABLE employees
DROP COLUMN phone,
DROP COLUMN linkedin_url;

-- PostgreSQL: DROP with dependency handling
ALTER TABLE employees DROP COLUMN dept_id CASCADE;
```

### Modify Column (Change Data Type / Constraints)
```sql
-- MySQL
ALTER TABLE employees
MODIFY COLUMN salary DECIMAL(12, 2) NOT NULL;

-- PostgreSQL
ALTER TABLE employees
ALTER COLUMN salary TYPE DECIMAL(12, 2);

ALTER TABLE employees
ALTER COLUMN salary SET NOT NULL;

ALTER TABLE employees
ALTER COLUMN salary DROP NOT NULL;

ALTER TABLE employees
ALTER COLUMN salary SET DEFAULT 50000;

ALTER TABLE employees
ALTER COLUMN salary DROP DEFAULT;

-- SQL Server
ALTER TABLE employees
ALTER COLUMN salary DECIMAL(12, 2) NOT NULL;
```

### Rename Column
```sql
-- MySQL 8.0+
ALTER TABLE employees
RENAME COLUMN first_name TO fname;

-- PostgreSQL
ALTER TABLE employees
RENAME COLUMN first_name TO fname;

-- SQL Server
EXEC sp_rename 'employees.first_name', 'fname', 'COLUMN';
```

### Rename Table
```sql
-- MySQL / PostgreSQL
ALTER TABLE employees RENAME TO staff;

-- SQL Server
EXEC sp_rename 'employees', 'staff';
```

### Add Constraint
```sql
-- Add PRIMARY KEY
ALTER TABLE employees
ADD CONSTRAINT pk_employees PRIMARY KEY (id);

-- Add FOREIGN KEY
ALTER TABLE employees
ADD CONSTRAINT fk_emp_dept
FOREIGN KEY (dept_id) REFERENCES departments(id)
ON DELETE SET NULL
ON UPDATE CASCADE;

-- Add UNIQUE constraint
ALTER TABLE employees
ADD CONSTRAINT uq_email UNIQUE (email);

-- Add CHECK constraint
ALTER TABLE employees
ADD CONSTRAINT chk_salary CHECK (salary >= 0);

-- Add DEFAULT (column-level, not named constraint)
ALTER TABLE employees
ALTER COLUMN is_active SET DEFAULT TRUE;
```

### Drop Constraint
```sql
-- MySQL
ALTER TABLE employees DROP PRIMARY KEY;
ALTER TABLE employees DROP FOREIGN KEY fk_emp_dept;
ALTER TABLE employees DROP INDEX uq_email;
ALTER TABLE employees DROP CHECK chk_salary;

-- PostgreSQL / SQL Server
ALTER TABLE employees DROP CONSTRAINT pk_employees;
ALTER TABLE employees DROP CONSTRAINT fk_emp_dept;
ALTER TABLE employees DROP CONSTRAINT uq_email;
ALTER TABLE employees DROP CONSTRAINT chk_salary;
```

### Enable / Disable Constraint (SQL Server / Oracle)
```sql
-- SQL Server: disable
ALTER TABLE employees NOCHECK CONSTRAINT fk_emp_dept;
-- re-enable
ALTER TABLE employees CHECK CONSTRAINT fk_emp_dept;

-- PostgreSQL: disable triggers (which enforce some constraints)
ALTER TABLE employees DISABLE TRIGGER ALL;
ALTER TABLE employees ENABLE TRIGGER ALL;
```

## 3. DROP TABLE

```sql
-- Basic drop
DROP TABLE table_name;

-- Conditional drop (no error if table doesn't exist)
DROP TABLE IF EXISTS table_name;

-- Drop multiple tables
DROP TABLE IF EXISTS table1, table2, table3;

-- Drop with cascade (PostgreSQL) — drops dependent objects too
DROP TABLE employees CASCADE;

-- Drop with restrict (default) — fails if dependents exist
DROP TABLE employees RESTRICT;
```

## 4. TRUNCATE TABLE

Removes all rows from a table but keeps the structure.

```sql
-- Basic truncate
TRUNCATE TABLE employees;

-- PostgreSQL: with cascade (also truncates referencing tables)
TRUNCATE TABLE departments CASCADE;

-- PostgreSQL: restart identity (resets auto-increment)
TRUNCATE TABLE employees RESTART IDENTITY;

-- PostgreSQL: combined
TRUNCATE TABLE employees RESTART IDENTITY CASCADE;

-- SQL Server
TRUNCATE TABLE employees;
```

### TRUNCATE vs DELETE vs DROP

| Feature | TRUNCATE | DELETE | DROP |
|---------|----------|--------|------|
| Removes rows | All rows | Selected rows | Entire table |
| WHERE clause | No | Yes | N/A |
| Can rollback | No (usually) | Yes | No |
| Triggers fired | No | Yes | No |
| Resets identity | Yes | No | N/A |
| Performance | Fast | Slow (logged) | Fast |
| Keeps structure | Yes | Yes | No |

## 5. RENAME

```sql
-- MySQL
RENAME TABLE old_name TO new_name;

-- Rename multiple tables at once (MySQL)
RENAME TABLE
    employees TO staff,
    departments TO divisions;

-- PostgreSQL
ALTER TABLE old_name RENAME TO new_name;

-- SQL Server
EXEC sp_rename 'old_name', 'new_name';
```

## 6. CREATE DATABASE / SCHEMA

### Create Database
```sql
-- MySQL
CREATE DATABASE company_db;
CREATE DATABASE IF NOT EXISTS company_db;
CREATE DATABASE company_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- PostgreSQL
CREATE DATABASE company_db;
CREATE DATABASE company_db
    OWNER = admin_user
    ENCODING = 'UTF8'
    LC_COLLATE = 'en_US.UTF-8'
    LC_CTYPE = 'en_US.UTF-8';

-- SQL Server
CREATE DATABASE company_db;
```

### Drop Database
```sql
DROP DATABASE company_db;
DROP DATABASE IF EXISTS company_db;
```

### Use Database (MySQL)
```sql
USE company_db;
```

### Create Schema (PostgreSQL / SQL Server)
```sql
CREATE SCHEMA hr;
CREATE SCHEMA IF NOT EXISTS hr;

-- Create table in schema
CREATE TABLE hr.employees (
    id   INT PRIMARY KEY,
    name VARCHAR(100)
);

-- Query with schema prefix
SELECT * FROM hr.employees;

-- Drop schema
DROP SCHEMA hr CASCADE;
```

## 7. CREATE INDEX

```sql
-- Basic index (see Indexes file for full coverage)
CREATE INDEX idx_employees_email ON employees (email);

-- Unique index
CREATE UNIQUE INDEX uq_employees_email ON employees (email);

-- Composite index
CREATE INDEX idx_employees_name ON employees (last_name, first_name);

-- Drop index
DROP INDEX idx_employees_email;                            -- PostgreSQL
DROP INDEX idx_employees_email ON employees;               -- MySQL
DROP INDEX idx_employees_email ON dbo.employees;           -- SQL Server
```

## 8. CREATE SEQUENCE

Sequences generate unique numeric values, typically used for primary keys.

### PostgreSQL
```sql
-- Create sequence
CREATE SEQUENCE employee_id_seq
    START WITH 1
    INCREMENT BY 1
    NO MINVALUE
    NO MAXVALUE
    CACHE 1;

-- Use sequence
INSERT INTO employees (id, name)
VALUES (nextval('employee_id_seq'), 'Alice');

-- Get current value
SELECT currval('employee_id_seq');

-- Advance without using value
SELECT nextval('employee_id_seq');

-- Alter sequence
ALTER SEQUENCE employee_id_seq RESTART WITH 100;
ALTER SEQUENCE employee_id_seq INCREMENT BY 5;

-- Drop sequence
DROP SEQUENCE employee_id_seq;
```

### MySQL (8.0+)
MySQL doesn't have sequences natively; use `AUTO_INCREMENT` or a workaround:
```sql
-- Workaround using a dedicated table
CREATE TABLE sequences (
    name    VARCHAR(50) PRIMARY KEY,
    current_val INT NOT NULL
);
INSERT INTO sequences VALUES ('employee_id', 0);

-- Get next value
UPDATE sequences SET current_val = (@next := current_val + 1) WHERE name = 'employee_id';
SELECT @next;
```

## 9. Temporary Tables

Temporary tables exist only for the duration of a session or transaction.

### Session-Scoped Temporary Tables
```sql
-- MySQL / SQL Server
CREATE TEMPORARY TABLE temp_sales (
    id      INT,
    amount  DECIMAL(10,2),
    region  VARCHAR(50)
);

-- Populate
INSERT INTO temp_sales SELECT id, amount, region FROM sales WHERE sale_date = CURRENT_DATE;

-- Use
SELECT region, SUM(amount) FROM temp_sales GROUP BY region;

-- Automatically dropped when session ends
-- Or explicitly:
DROP TEMPORARY TABLE temp_sales;  -- MySQL
DROP TABLE #temp_sales;           -- SQL Server (# prefix for local temp)
DROP TABLE ##temp_sales;          -- SQL Server (## prefix for global temp)
```

### PostgreSQL Temporary Tables
```sql
-- Dropped at end of session (default)
CREATE TEMP TABLE temp_sales (
    id     INT,
    amount DECIMAL(10,2)
);

-- Dropped at end of transaction
CREATE TEMP TABLE temp_sales (
    id     INT,
    amount DECIMAL(10,2)
) ON COMMIT DROP;

-- Preserved but empty after transaction
CREATE TEMP TABLE temp_sales (
    id     INT,
    amount DECIMAL(10,2)
) ON COMMIT DELETE ROWS;
```

### CTEs vs Temp Tables
```sql
-- CTE (exists only within one query)
WITH temp_data AS (
    SELECT id, amount FROM sales WHERE sale_date = CURRENT_DATE
)
SELECT * FROM temp_data;

-- Temp table (persists across multiple queries in the session)
CREATE TEMP TABLE temp_data AS
    SELECT id, amount FROM sales WHERE sale_date = CURRENT_DATE;

SELECT * FROM temp_data;  -- Usable in next statement
```

## 10. COMMENT

### Column and Table Comments
```sql
-- MySQL: inline comment
CREATE TABLE employees (
    id    INT     COMMENT 'Primary key, auto-incremented',
    name  VARCHAR(100) COMMENT 'Full legal name'
) COMMENT = 'Stores all employee records';

-- Add/update comment after creation
ALTER TABLE employees COMMENT = 'Employee master data';
ALTER TABLE employees MODIFY COLUMN id INT COMMENT 'Unique employee identifier';

-- PostgreSQL: COMMENT ON
COMMENT ON TABLE employees IS 'Stores all employee records';
COMMENT ON COLUMN employees.id IS 'Primary key, auto-incremented';
COMMENT ON COLUMN employees.salary IS 'Annual salary in USD';

-- View comments (PostgreSQL)
SELECT
    c.column_name,
    pgd.description
FROM information_schema.columns c
LEFT JOIN pg_catalog.pg_statio_all_tables st ON st.relname = c.table_name
LEFT JOIN pg_catalog.pg_description pgd ON pgd.objoid = st.relid
    AND pgd.objsubid = c.ordinal_position
WHERE c.table_name = 'employees';
```

## DDL Quick Reference

```sql
-- Create
CREATE TABLE t (col datatype [constraint], ...);
CREATE TABLE IF NOT EXISTS t (...);
CREATE TABLE t AS SELECT ...;
CREATE DATABASE db;
CREATE SCHEMA s;
CREATE INDEX idx ON t(col);
CREATE SEQUENCE seq START 1 INCREMENT 1;

-- Alter table
ALTER TABLE t ADD COLUMN col datatype;
ALTER TABLE t DROP COLUMN col;
ALTER TABLE t MODIFY COLUMN col new_type;     -- MySQL
ALTER TABLE t ALTER COLUMN col TYPE new_type; -- PostgreSQL
ALTER TABLE t RENAME COLUMN old TO new;
ALTER TABLE t RENAME TO new_name;
ALTER TABLE t ADD CONSTRAINT name ...;
ALTER TABLE t DROP CONSTRAINT name;

-- Drop
DROP TABLE t;
DROP TABLE IF EXISTS t;
DROP TABLE t CASCADE;
DROP DATABASE db;
DROP SCHEMA s CASCADE;
DROP INDEX idx;
DROP SEQUENCE seq;

-- Truncate
TRUNCATE TABLE t;
TRUNCATE TABLE t RESTART IDENTITY CASCADE;  -- PostgreSQL
```
